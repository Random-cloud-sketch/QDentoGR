CREATE TABLE IF NOT EXISTS medical_history (rowid INTEGER NOT NULL PRIMARY KEY, patient_rowid INTEGER NOT NULL REFERENCES patient (rowid) ON DELETE CASCADE ON UPDATE CASCADE, dentist_rowid INTEGER REFERENCES dentist (rowid) ON DELETE SET NULL ON UPDATE CASCADE, saved TEXT NOT NULL, history_date TEXT NOT NULL, general_health INTEGER NOT NULL DEFAULT 0, physician TEXT, physician_contact TEXT, allergy_reaction TEXT, smoking INTEGER NOT NULL DEFAULT 0, cigarettes_per_day INTEGER, smoking_years INTEGER, smoking_quit TEXT, lifestyle_other TEXT, oral_hygiene TEXT, dental_other TEXT, notes TEXT);
CREATE INDEX IF NOT EXISTS medical_history_patient ON medical_history (patient_rowid);
CREATE TABLE IF NOT EXISTS medical_history_item (medical_history_rowid INTEGER NOT NULL REFERENCES medical_history (rowid) ON DELETE CASCADE ON UPDATE CASCADE, item TEXT NOT NULL, answer INTEGER NOT NULL, details TEXT, PRIMARY KEY (medical_history_rowid, item));
CREATE TABLE IF NOT EXISTS medical_history_medication (rowid INTEGER NOT NULL PRIMARY KEY, medical_history_rowid INTEGER NOT NULL REFERENCES medical_history (rowid) ON DELETE CASCADE ON UPDATE CASCADE, name TEXT NOT NULL, dose TEXT, frequency TEXT, indication TEXT, notes TEXT);
CREATE TABLE IF NOT EXISTS medical_history_surgery (rowid INTEGER NOT NULL PRIMARY KEY, medical_history_rowid INTEGER NOT NULL REFERENCES medical_history (rowid) ON DELETE CASCADE ON UPDATE CASCADE, surgery TEXT NOT NULL, date TEXT, reason TEXT, hospital TEXT, notes TEXT);
