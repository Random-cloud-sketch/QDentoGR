CREATE TABLE IF NOT EXISTS recall (patient_rowid INTEGER PRIMARY KEY NOT NULL REFERENCES patient (rowid) ON DELETE CASCADE ON UPDATE CASCADE, active INTEGER NOT NULL DEFAULT 1, next_date TEXT, interval_months INTEGER, last_date TEXT, notes TEXT);
CREATE TABLE IF NOT EXISTS recall_history (rowid INTEGER PRIMARY KEY, patient_rowid INTEGER NOT NULL REFERENCES patient (rowid) ON DELETE CASCADE ON UPDATE CASCADE, time TEXT NOT NULL, event TEXT NOT NULL, value TEXT, previous_value TEXT, appointment_rowid INTEGER, note TEXT);
CREATE INDEX IF NOT EXISTS recall_history_patient ON recall_history (patient_rowid, time);
ALTER TABLE appointment ADD COLUMN recall INTEGER NOT NULL DEFAULT 0;
ALTER TABLE appointment ADD COLUMN recall_status TEXT;
CREATE TRIGGER IF NOT EXISTS recall_appointment_booked AFTER INSERT ON appointment WHEN NEW.recall = 1 AND NEW.patient_rowid IS NOT NULL BEGIN INSERT INTO recall_history (patient_rowid, time, event, value, previous_value, appointment_rowid) VALUES (NEW.patient_rowid, strftime('%Y-%m-%dT%H:%M:%S', 'now', 'localtime'), 'booked', NEW.start, NULL, NEW.rowid); END;
CREATE TRIGGER IF NOT EXISTS recall_appointment_marked AFTER UPDATE OF recall ON appointment WHEN OLD.recall = 0 AND NEW.recall = 1 AND NEW.patient_rowid IS NOT NULL BEGIN INSERT INTO recall_history (patient_rowid, time, event, value, previous_value, appointment_rowid) VALUES (NEW.patient_rowid, strftime('%Y-%m-%dT%H:%M:%S', 'now', 'localtime'), 'booked', NEW.start, NULL, NEW.rowid); END;
CREATE TRIGGER IF NOT EXISTS recall_appointment_unmarked AFTER UPDATE OF recall ON appointment WHEN OLD.recall = 1 AND NEW.recall = 0 AND OLD.patient_rowid IS NOT NULL BEGIN INSERT INTO recall_history (patient_rowid, time, event, value, previous_value, appointment_rowid) VALUES (OLD.patient_rowid, strftime('%Y-%m-%dT%H:%M:%S', 'now', 'localtime'), 'unmarked', OLD.start, NULL, OLD.rowid); END;
CREATE TRIGGER IF NOT EXISTS recall_appointment_rescheduled AFTER UPDATE OF start ON appointment WHEN OLD.recall = 1 AND NEW.recall = 1 AND NEW.patient_rowid IS NOT NULL AND OLD.start <> NEW.start BEGIN INSERT INTO recall_history (patient_rowid, time, event, value, previous_value, appointment_rowid) VALUES (NEW.patient_rowid, strftime('%Y-%m-%dT%H:%M:%S', 'now', 'localtime'), 'rescheduled', NEW.start, OLD.start, NEW.rowid); END;
CREATE TRIGGER IF NOT EXISTS recall_appointment_cancelled AFTER DELETE ON appointment WHEN OLD.recall = 1 AND OLD.patient_rowid IS NOT NULL AND IFNULL(OLD.recall_status, '') = '' BEGIN INSERT INTO recall_history (patient_rowid, time, event, value, previous_value, appointment_rowid) VALUES (OLD.patient_rowid, strftime('%Y-%m-%dT%H:%M:%S', 'now', 'localtime'), 'cancelled', OLD.start, NULL, OLD.rowid); END;
